\begin{conf-abstract}
{Evaluating the effects of multiple variable on the dissolution of \ce{CO2}}
{Mohammed Alshareef}
{University of Oxford}
{In a tube setup we simulate the injection of \ce{CO2} through a column of water in order to assess the affects of bubble size and flux on the dossolution of \ce{CO2}. A review of the of data collected by the miniRUEDI through the headspace within the column is then used to identify the signature of \ce{CO2} leakage.}
\end{conf-abstract}
